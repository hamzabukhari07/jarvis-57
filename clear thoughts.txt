## 🔧 Architecture Integration: The ZEZO Voice Hub
To implement this without breaking your clear module boundaries, you can house Pipecat inside a voice/ or telemetry/ module. Below is a structural example of how Pipecat maps to your Gemini Live + Function Dispatcher setup:

import asynciofrom pipecat.frames.frames import AudioRawFrame, TextFramefrom pipecat.pipeline.pipeline import Pipelinefrom pipecat.services.google import GoogleLLMServicefrom pipecat.transports.services.livekit import LiveKitTransportfrom zezo.core.dispatcher import zezo_dispatcher  # Your Dispatcher
async def run_zezo_voice_loop():
    # 1. Transport Layer (Handles Mic/Audio Room)
    transport = LiveKitTransport()

    # 2. Gemini Live Connection (Natively supports voice config & system prompts)
    gemini_live = GoogleLLMService(
        api_key="YOUR_FREE_GEMINI_KEY",
        model="gemini-2.5-flash",
        voice="Aoede",
        system_instruction="You are ZEZO OS, an autonomous desktop AI. You have tools available."
    )

    # 3. Connect Gemini Live Function Calling directly to ZEZO's tool dispatcher
    @gemini_live.on_function_call
    async def handle_tool_call(function_name, arguments):
        # Hooks directly into your "Action Executor (Risk Gated)" block
        result = await zezo_dispatcher.execute(function_name, arguments)
        return result

    # 4. Orchestrate into a single non-blocking pipeline
    pipeline = Pipeline([
        transport.input(),    # Listens to mic
        gemini_live,          # Brain / Conversational loop
        transport.output()   # Speaks back out
    ])

    await pipeline.run()
if __name__ == "__main__":
    asyncio.run(run_zezo_voice_loop())
